Display_todo_item

alias: Display_todo_list
description: ''
triggers:
  - trigger: time_pattern
    seconds: /15
conditions: []
actions:
  - action: todo.get_items
    metadata: {}
    target:
      entity_id: todo.work_list
    data:
      status: needs_action
    response_variable: Work_List_items
  - variables:
      grocery_text: >-
        {% set items = Work_List_items['todo.work_list']['items'] | default([])
        %} {% if items | length > 0 %}
          {{ '• ' ~ (items | map(attribute='summary') | list | join('\n• ')) }}
        {% else %}
          Empty
        {% endif %}
  - if:
      - condition: template
        value_template: |
          {{ grocery_text | trim != states('text.mywatch_work_text') | trim }}
    then:
      - action: text.set_value
        target:
          entity_id:
            - text.round146_work_text
            - text.mywatch_work_text
        data:
          value: '{{ grocery_text | trim }}'
mode: single
